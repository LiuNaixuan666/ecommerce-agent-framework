            <div style={{ flex: 1, overflowY: 'auto', padding: '1.5rem', display: 'flex', flexDirection: 'column', gap: '1rem' }}>
              {currentConversation.messages.map(message => (
                <div
                  key={message.id}
                  style={{
                    display: 'flex',
                    justifyContent: message.isUser ? 'flex-end' : 'flex-start',
                  }}
                >
                  <div className={`message ${message.isUser ? 'message-user' : 'message-bot'}`}>
                    <p style={{ margin: 0, whiteSpace: 'pre-wrap', wordBreak: 'break-word' }}>{message.content}</p>
                    <p style={{ margin: '0.5rem 0 0', fontSize: '0.78rem', opacity: 0.75 }}>{message.timestamp.toLocaleTimeString('zh-CN', { hour12: false })}</p>
                  </div>
                </div>
              ))}

              {isLoading && (
                <div style={{ display: 'flex', justifyContent: 'flex-start' }}>
                  <div className="message message-bot" style={{ backgroundColor: '#f1f5f9' }}>
                    <div style={{ display: 'flex', alignItems: 'center', gap: '0.6rem' }}>
                      <div className="animate-spin" style={{ width: '1rem', height: '1rem', borderTop: '2px solid #475569' }} />
                      <span style={{ color: '#475569' }}>AI 正在生成回答...</span>
                    </div>
                  </div>
                </div>
              )}

              <div ref={messagesEndRef} />
            </div>

            <div style={{ padding: '1rem 1.5rem 1.5rem', backgroundColor: '#f8fafc', borderTop: '1px solid #e2e8f0' }}>
              <div style={{ display: 'flex', gap: '0.75rem' }}>
                <input
                  type="text"
                  value={inputMessage}
                  onChange={e => setInputMessage(e.target.value)}
                  onKeyPress={handleKeyPress}
                  placeholder="输入您的问题，例如：退货政策是什么？"
                  className="input"
                  style={{ flex: 1 }}
                  disabled={isLoading}
                />
                <button
                  onClick={sendMessage}
                  disabled={isLoading || !inputMessage.trim()}
                  className="btn btn-primary"
                  style={{ padding: '0.95rem 1.2rem' }}
                >
                  <Send size={18} />
                </button>
              </div>
            </div>
          </>
        ) : (
          <div style={{ flex: 1, display: 'flex', alignItems: 'center', justifyContent: 'center', padding: '2rem' }}>
            <div style={{ textAlign: 'center', color: '#475569' }}>
              <MessageCircle size={56} style={{ color: '#cbd5e1', marginBottom: '1rem' }} />
              <h3 style={{ margin: 0, fontSize: '1.25rem', fontWeight: 700 }}>开始一个新的对话</h3>
              <p style={{ margin: '0.75rem 0 1.25rem', lineHeight: 1.7 }}>上传知识文档并发送消息，AI 将为你提供智能客服答复。</p>
              <button
                onClick={createNewConversation}
                className="btn btn-primary"
                style={{ padding: '0.95rem 1.4rem' }}
              >
                新建会话
              </button>
            </div>
          </div>
        )}
      </div>

      {notification && (
        <div style={{
          position: 'fixed',
          right: '1.5rem',
          bottom: '1.5rem',
          backgroundColor: '#111827',
          color: '#ffffff',
          padding: '0.95rem 1rem',
          borderRadius: '1rem',
          boxShadow: '0 15px 30px rgba(15, 23, 42, 0.18)',
          zIndex: 20,
        }}>
          {notification}
        </div>
      )}
    </div>
  )
}
